#and there is decorator

def CommandStatusWrapper(callbacks:Dict[str, Callable] = {}):
    def wrapper(fn:Callable):
        def exec(machine:object ,*args, **kwargs):
            res:ActionResult = fn(machine, *args, **kwargs)
            if res.status == "ok":
                return True
            if res.status in callbacks:
                return callbacks[res.status](machine, *args, **kwargs)
            notify(f"{machine.get_id()} was failed called {fn}. result {res.status} with message {res.message}")
            return False
        return exec
    return wrapper

class BioExchangeMachine(BaseMachine):
    machine:BioExchange
    comms = comms("bio.orders", BioExchangeOrder)
    inv = get_component("inventory")
    
    def __init__(self, machine:BioExchange):
        super().__init__(machine)

    @staticmethod
    def CollectRequiresForOrder(order:BioOrder) -> Dict[str, int]:
        requires = {}
        for k,v in order.requires.items():
            requires[k] = v
        for k,v in order.delivered.items():
            requires[k] -= v
        for k,v in order.in_transit.items():
            requires[k] -= v
        return requires
        
    def CheckRequested(self, order:BioOrder):
        queried:List[BioExchangeOrder] = [msg for msg in self.comms.search(lambda msg: msg.requested_by == self.machine.id and msg.order_id == order.id)]
        
        requires = BioExchangeMachine.CollectRequiresForOrder(order)
        for k, v in requires.items(): 
            if not any(k == q_one.required for q_one in queried):
                self.comms.send(BioExchangeOrder(
                    requested_by = self.machine.id,
                    order_id = order.id,
                    required = k,
                    count = v
                ))
               
    def postInit(self):
        self.machine.input.connect("inventory")
        if order := self.machine.active_order():
            self.CheckRequested(order)

    def OrderWasCompletedCallback(self):
        notify("Bio exchange complete order")
        return True

class Point:
    def __init__(self, _pos):
        if isinstance(_pos, (int, float)):
            self.x = self.y = _pos
        elif isinstance(_pos, (tuple, list)):
            self.x, self.y = _pos
        
        
        elif isinstance(_pos, str):
            c = get_component(_pos)
            if c:
                self.x, self.y = c.coords()
            else:
                for site in get_component("journal").discovered_sites("nocturna"):
                    if site.id == _pos:
                        self.x, self.y = site.position()
                        return
                raise TypeError(f"Cannot create Point from {_pos}")
        # if isinstance(_pos, str):
        #     self.x = ord(_pos[0]) - ord("A")
        #     self.y = int(_pos[1:])
            #TODO: raise ValueError
        elif hasattr(_pos, "x") and hasattr(_pos, "y"):
            self.x = _pos.x
            self.y = _pos.y
        else:
            raise TypeError(f"Cannot create Point from {type(_pos)}")


    def nav_mul(self, nav_modules_count):
        return 1 + 1.25 * nav_modules_count + 0.35 * nav_modules_count ** 2
        
    #TODO: ADD DRAW CALCS FOR GO TASK ENERGY SIMULATE AND RESOLVE
    def calc_travel_power_static(self, from_pos:Point, 
                                 to_pos:Point, 
                                 throttle:float,
                                 nav_modules_count:int, 
                                 active_modules:int,
                                 cargo:int,
                                 speed:float
                                ):
        distance = from_pos.distance(to_pos)
        if distance == 0:
            return 0
        nav_mul = self.nav_mul(nav_modules_count) 
        

        draw = (3 + 8 * active_modules + 0.04 * cargo) * (throttle ** 1.5) * nav_mul
        pwr = draw/(speed/100)

        
        return distance/100 * pwr 

    def calc_travel_throttle_static(self,
                                    from_pos: Point,
                                    to_pos: Point,
                                    energy: float,
                                    nav_modules_count: int,
                                    active_modules: int,
                                    cargo: int,
                                ):
        distance = from_pos.distance(to_pos)
    
        if distance == 0:
            return 0.0  
        base = 3 + 8 * active_modules + 0.04 * cargo
        speed_mul = self.nav().speed_multiplier()
        k = distance / 100 * base * self.nav_mul(nav_modules_count) / speed_mul
        throttle = (energy / k) ** 2
        return min(1.0, max(0.0, throttle))

    def count_active_modules(self):
        count = 0
        for module in self.parent.modules():
            if module.module_id and any(module.module_id.startswith(key) for key in ["nav_module", "drill_module", "sonar_module", "constructor_module"]):
                count += 1
        return count

class InventoryData:
    notebook = get_component("notebook")
    n_k = "inventory"

    def __init__(self):
        if not self.notebook.has("inventory"):
            self.notebook.set(self.n_k, InventoryEntity().dump())

    def transaction(self, callback:Callable):
        self.notebook.transaction(self.n_k, InventoryEntity().dump(), callback)

    def data(self) -> InventoryEntity:
        return InventoryEntity(**self.notebook.get(self.n_k))

    def AddOutpost(self, outpost_id) -> ActionResult|None:
        if outpost_id in self.data().outposts:
            return None
        def transaction(entry):
            return InventoryEntity(**entry).add_outpost(outpost_id).dump()
        return self.transaction(transaction)

    def FindOutpost(self, outpost_id) -> OutpostInventory | None:
        return self.data().outposts.get(outpost_id, None)

class RConsole:
    console = get_component("console")
    def __init__(self):
        self.indentation_level = 0
        self.indentation_colors: dict[int, str] = {}

    def print(self, msg: str, color: Optional[str] = None):
        out = "".join(["┃   "] * self.indentation_level) + msg
        outColor = self.indentation_colors.get(self.indentation_level, "#FFFFFF") if color is None else color
        self.console.print(out, level="info", color=outColor, timestamp=True)

    def start(self, msg: str, color: str = "#FFFFFF"):
        level = self.indentation_level
        self.indentation_colors[level] = color
        self.indentation_colors[level + 1] = color
        self.print("┏━ " + msg)
        self.indentation_level += 1

    def end(self, msg: str, color: Optional[str] = None):
        self.indentation_level -= 1
        self.print("┗━ " + msg, color)
        level = self.indentation_level
        self.indentation_colors.pop(level + 1)
        self.indentation_colors.pop(level)

def pretty_format(obj, indent=0, in_prefix = ""):
    lines = []
    prefix = in_prefix + " " * indent

    if isinstance(obj, dict):
        if len(obj) > 0:
            lines.append(prefix + "{")

            for key, value in obj.items():
                if isinstance(value, (list, tuple, set, dict)):
                    formatted = pretty_format(value, indent + 4, in_prefix = in_prefix)
                    value_lines = formatted.split("\n")
                    
                    first_line = value_lines[0].strip()
                    lines.append(prefix + f"    {key!r}: " + first_line)
                    lines.extend(value_lines[1:])
                else:
                    lines.append(prefix + f"    {key!r}: {repr(value)}")

            lines.append(prefix + "}")
        else:
            lines.append(prefix + "{ }")

    elif isinstance(obj, (list, tuple, set)):
        brackets = {
            "list": ("[", "]"),
            "tuple": ("(", ")"),
            "set": ("{", "}"),
        }[type(obj)]

        if len(obj) > 0:
            lines.append(prefix + brackets[0])

            for value in obj:
                lines.append(pretty_format(value, indent + 4, in_prefix = in_prefix))

            lines.append(prefix + brackets[1])
        else:
            lines.append(prefix + brackets[0] + " " + brackets[1])

    else:
        lines.append(prefix + repr(obj))

    return "\n".join(lines)

class RConsole:

        return self.block_as(lambda name, *args, **kwargs: name)(fn)

        def decorator(fn):
            @_functools.wraps(fn)
            def wrapped(*args, **kwargs):
                return self.run(
                    make_label(fn.__name__, *args, **kwargs),
                    lambda: fn(*args, **kwargs),
                )
            return wrapped
        return decorator

    def run(self, label: str, fn):
        self.begin(label)
        try:
            ret = fn()
        except Exception as e:
            self.end(f"{label} !! {e!r}")
            raise
        self.end(label)
        return ret

class Overload:

    def __init__(self):
        # print("init")
        self.functions = {}

    def add(self, *types):
        # print(f"add {types}")
        def decorator(function):
            types_ingame = tuple(type.__name__ for type in types)
            self.functions[types_ingame] = function
            return function

        return decorator

    def call(self, instance, *args):
        # print(f"call {instance} {args} self {self.functions}")
        signature = tuple(type(x) for x in args)

        function = self.functions.get(signature)

        if function is None:
            raise TypeError(
                "No matching overload: " + str(signature)
            )

        return function(instance, *args)

def CommandStatusWrapper(callbacks:Dict[str, Callable] = {}):
    def wrapper(fn:Callable):
        def exec(machine:object ,*args, **kwargs):
            res:ActionResult = fn(machine, *args, **kwargs)
            if res.status == "ok":
                return True
            if res.status in callbacks:
                return callbacks[res.status](machine, *args, **kwargs)
            notify(f"{machine.get_id()} was failed called {fn}. result {res.status} with message {res.message}")
            return False
        return exec
    return wrapper

    def CollectRequiresForOrder(order:BioOrder) -> Dict[str, int]:
        requires = {}
        for k,v in order.requires.items():
            requires[k] = v
        for k,v in order.delivered.items():
            requires[k] -= v
        for k,v in order.in_transit.items():
            requires[k] -= v
        return requires

                    requested_by = self.machine.id,
                    order_id = order.id,
                    required = k,
                    count = v
